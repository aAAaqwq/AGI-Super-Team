## 五、可复用代码：回撤档位回测

```python
import urllib.request, json, statistics as st

def fetch(u, t=30, tries=3):
    for i in range(tries):
        try:
            return urllib.request.urlopen(urllib.request.Request(
                u, headers={"User-Agent":"Mozilla/5.0"}), timeout=t).read().decode('utf-8','ignore')
        except Exception:
            if i == tries-1: raise

# 周线（600 周 ≈ 11.5 年）
k = json.loads(fetch("https://web.ifzq.gtimg.cn/appstock/app/fqkline/get?param=sh510300,week,,,600,qfq"))
rows = k['data']['sh510300'].get('qfqweek') or k['data']['sh510300'].get('week')
wc = [float(r[2]) for r in rows]
n = len(wc)

# 峰值回撤
peak, dd = wc[0], []
for p in wc:
    peak = max(peak, p)
    dd.append(p/peak - 1)

# 按档位回测
for lo, hi in [(-0.15,-0.10), (-0.25,-0.15), (-0.35,-0.25), (-0.50,-0.35)]:
    sig = [i for i in range(60, n) if lo <= dd[i] < hi]
    # 去重：信号间隔 > 8 周
    dedup = []
    for i in sig:
        if not dedup or i - dedup[-1] > 8: dedup.append(i)
    for w in (13, 26, 52):
        rs = [wc[i+w]/wc[i]-1 for i in dedup if i+w < n]
        if rs:
            win = sum(1 for x in rs if x > 0)/len(rs)
            print(f"{lo:.0%}~{hi:.0%} {w}周 胜率{win:.0%} 中位{st.median(rs):+.1%}")
```

## 六、可复用代码：风险收益时间比

```python
import math, statistics as st

rets = [c[i]/c[i-1]-1 for i in range(1, len(c))]
mean = sum(rets)/len(rets); sd = st.stdev(rets)
ann = mean*250; vol = sd*math.sqrt(250)
sharpe = (ann-0.02)/vol

peak, mdd = c[0], 0
for p in c:
    peak = max(peak, p); mdd = min(mdd, p/peak-1)

downs = [min(x,0) for x in rets]
dsd = math.sqrt(sum(x*x for x in downs)/len(downs))*math.sqrt(250)

print(f"年化 {ann:.2%} 波动 {vol:.2%} 夏普 {sharpe:.2f}")
print(f"最大回撤 {mdd:.2%} 索提诺 {(ann-0.02)/dsd:.2f} 卡玛 {abs(ann/mdd):.2f}")
```
